// Audio Context Singleton
let audioCtx = null;

function getAudioContext() {
  if (!audioCtx) {
    const AudioContextClass = window.AudioContext || window.webkitAudioContext;
    audioCtx = new AudioContextClass();
  }
  if (audioCtx.state === 'suspended') {
    audioCtx.resume();
  }
  return audioCtx;

  const btn = document.getElementById('btn-tanpura-toggle');

    isTanpuraRunning = true;
    btn.textContent = 'STOP DRONE';

function setTanpuraVolume(val) {

function pluckNextTanpuraString() {
  if (!isTanpuraRunning) return;
  const ctx = getAudioContext();
  const pitch = tanpuraPitches[currentString];
  
  // Pluck individual string with non-linear Jawari overtones
  const osc = ctx.createOscillator();
  const oscGain = ctx.createGain();
  
  // Complex sawtooth with slight detune creates authentic thread buzzing
  osc.type = 'sawtooth';
  osc.frequency.setValueAtTime(pitch.freq, ctx.currentTime);
  
  // Low-pass filter simulating the curved wooden bridge (Jawari)
  const filter = ctx.createBiquadFilter();
  filter.type = 'lowpass';
  filter.frequency.setValueAtTime(pitch.freq * 12, ctx.currentTime);
  filter.frequency.exponentialRampToValueAtTime(pitch.freq * 2.5, ctx.currentTime + 3.5);
  filter.Q.setValueAtTime(4.0, ctx.currentTime);
  
  // Envelope
  oscGain.gain.setValueAtTime(0.001, ctx.currentTime);
  oscGain.gain.linearRampToValueAtTime(0.3, ctx.currentTime + 0.08);
  oscGain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 4.0);
  
  osc.connect(filter);
  filter.connect(oscGain);
  oscGain.connect(tanpuraMasterGain);
  
  osc.start(ctx.currentTime);
  osc.stop(ctx.currentTime + 4.2);
  
  currentString = (currentString + 1) % tanpuraPitches.length;
  tanpuraTimer = setTimeout(pluckNextTanpuraString, 1100);
}

    const dGain = ctx.createGain();
    dOsc.type = 'sine';
    dOsc.frequency.setValueAtTime(261.63, ctx.currentTime);

    dOsc.frequency.exponentialRampToValueAtTime(268.0, ctx.currentTime + 0.03);
    dOsc.frequency.exponentialRampToValueAtTime(261.63, ctx.currentTime + 0.3);
    
    dGain.gain.setValueAtTime(0.25, ctx.currentTime);
    dGain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.28);
    
    dOsc.connect(dGain);
    dGain.connect(ctx.destination);
    dOsc.start(ctx.currentTime);
    dOsc.stop(ctx.currentTime + 0.3);
  }
  
  // Synthesize Bayan (Bass drum with pitch modulation "Ghe" downward slide)
  if (beat.bayan) {
    const bOsc = ctx.createOscillator();
    const bGain = ctx.createGain();
    bOsc.type = 'triangle';
    
    // Pitch drops from 110Hz to 65Hz simulating the heel of the palm pressure
    bOsc.frequency.setValueAtTime(110.0, ctx.currentTime);
    bOsc.frequency.exponentialRampToValueAtTime(65.0, ctx.currentTime + 0.35);
    
    bGain.gain.setValueAtTime(0.4, ctx.currentTime);
    bGain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.4);
    
    bOsc.connect(bGain);
    bGain.connect(ctx.destination);
    bOsc.start(ctx.currentTime);
    bOsc.stop(ctx.currentTime + 0.42);
  }
  
  currentBeat = (currentBeat + 1) % 16;
  const intervalMs = (60.0 / tablaBpm) * 1000;
  tablaTimer = setTimeout(playNextTablaBeat, intervalMs);
}

function playSwara(name, freq, ratio, cents) {
  const ctx = getAudioContext();

  document.getElementById('current-swara').textContent = name;
  document.getElementById('current-freq').textContent = freq.toFixed(2) + ' Hz';
  document.getElementById('current-ratio').textContent = ratio;
  document.getElementById('current-cents').textContent = cents;
  
  // Continuous Monophonic Flute Voice with 140ms Meend portamento glide
  if (!bansuriOsc) {
    bansuriOsc = ctx.createOscillator();
    bansuriGain = ctx.createGain();
    
    // Triangle wave provides soft, hollow bamboo harmonics
    bansuriOsc.type = 'triangle';
    bansuriOsc.frequency.setValueAtTime(freq, ctx.currentTime);
    
    // 5.2 Hz Andolan vibrato LFO
    lfoOsc = ctx.createOscillator();
    const lfoGain = ctx.createGain();
    lfoOsc.frequency.setValueAtTime(5.2, ctx.currentTime);
    lfoGain.gain.setValueAtTime(2.5, ctx.currentTime); // ~12 cents modulation
    lfoOsc.connect(lfoGain);
    lfoGain.connect(bansuriOsc.frequency);
    lfoOsc.start(ctx.currentTime);
    
    // Envelope
    bansuriGain.gain.setValueAtTime(0.001, ctx.currentTime);
    bansuriGain.gain.linearRampToValueAtTime(0.35, ctx.currentTime + 0.08);
    
    bansuriOsc.connect(bansuriGain);
    bansuriGain.connect(ctx.destination);
    bansuriOsc.start(ctx.currentTime);
  } else {
    // Portamento Meend glide from last frequency to new frequency
    bansuriOsc.frequency.cancelScheduledValues(ctx.currentTime);
    bansuriOsc.frequency.setValueAtTime(lastFreq, ctx.currentTime);
    bansuriOsc.frequency.exponentialRampToValueAtTime(freq, ctx.currentTime + 0.14);
    
    // Refresh gain envelope
    bansuriGain.gain.cancelScheduledValues(ctx.currentTime);
    bansuriGain.gain.setValueAtTime(0.35, ctx.currentTime);
  }
  
  lastFreq = freq;
  
  // Gentle release if no subsequent note is pressed within 1.2 seconds
  bansuriGain.gain.setValueAtTime(0.35, ctx.currentTime + 0.8);
  bansuriGain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 1.4);
}

## 3. Mathematical & Cybernetic Foundations

### 3.1 The 22 Shrutis & Pure Harmonic Ratios
In Indian classical music, an octave is divided into 22 unequal microtonal intervals (*Shrutis*), derived through successive cycles of pure fifths (*Panchama-bhāva*, $3/2$) and pure fourths (*Madhyama-bhāva*, $4/3$), collapsed by octaves ($1/2$). This produces three fundamental microtonal step sizes:
1. **Pramāna Shruti** (Comma of Didymus): $\frac{81}{80} \approx 21.51\text{ cents}$
2. **Upabichitra / Nyūna Shruti** (Minor semitone): $\frac{25}{24} \approx 70.67\text{ cents}$
3. **Pūrna Shruti** (Major semitone): $\frac{16}{15} \approx 111.73\text{ cents}$

In Rāga Durgā (Bilawal Thāat), the pentatonic scale utilizes the following five microtonal coordinates:

$$\begin{aligned}
\text{Ṣaḍja (Sa)} &= \frac{1}{1} = 0.00\text{ cents} \quad (261.63\text{ Hz}) \\
\text{Triśruti Riṣabha (Re)} &= \frac{9}{8} = 203.91\text{ cents} \quad (294.33\text{ Hz}) \quad [\text{Samvādi}] \\
\text{Śuddha Madhyama (Ma)} &= \frac{4}{3} = 498.04\text{ cents} \quad (348.83\text{ Hz}) \\
\text{Pañcama (Pa)} &= \frac{3}{2} = 701.96\text{ cents} \quad (392.44\text{ Hz}) \\
\text{Catuḥśruti Dhaivata (Dha)} &= \frac{5}{3} = 884.36\text{ cents} \quad (436.04\text{ Hz}) \quad [\text{Vādi}]
\end{aligned}$$

Notice that the Vādi (dominant tone) is Chatushruti Dhaivata ($5/3$, $884.36\text{ cents}$), which is exactly 15.64 cents lower than Western 12-TET A4 (900 cents). This microtonal difference removes the acoustic beating that occurs in equal temperament, allowing the Bānsurī to merge flawlessly into the Tanpura's high harmonics.

### 3.2 Lyapunov Non-Chaotic Resonance
The acoustic interaction between the continuous flute trajectory $x(t)$ and the Tanpura drone harmonic series $y(t)$ is modeled as a non-linear dynamical system:

$$\frac{d}{dt} \begin{pmatrix} x \\ y \end{pmatrix} = \begin{pmatrix} -\gamma_1 & \omega_0 \\ -\omega_0 & -\gamma_2 \end{pmatrix} \begin{pmatrix} x \\ y \end{pmatrix} + \mathbf{F}_{\text{coupling}}(x, y)$$

Because $\lambda_{\max} < 0$, any transient phase disturbance or pitch deviation contracts exponentially toward the limit cycle attractor at rate $e^{-3.732 t}$.
