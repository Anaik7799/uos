(* gospel_dispatch_contracts.ml — Extended Gospel Contracts and Bounded Differential Oracles (EV-107) *)

type verdict =
  | Pass of { digest : string; timestamp : string }
  | FailClosed of { reason : string; error_code : int }

let contains_nul_byte s =
  String.contains s '\x00'

let contains_sql_injection s =
  let upper = String.uppercase_ascii s in
  let patterns = [
    "; DROP TABLE";
    "; DELETE FROM";
    "' OR '1'='1";
    "\" OR \"1\"=\"1";
    "UNION SELECT";
    "-- ";
    "/*";
  ] in
  List.exists (fun p ->
    try
      let _ = Str.search_forward (Str.regexp_string p) upper 0 in
      true
    with Not_found -> false
  ) patterns

let sha256_digest s =
  let h = Cryptokit.Hash.sha256 () in
  h#add_string s;
  Cryptokit.transform_string (Cryptokit.Hexa.encode ()) h#result

let get_iso_timestamp () =
  let t = Unix.gettimeofday () in
  let tm = Unix.gmtime t in
  Printf.sprintf "%04d-%02d-%02dT%02d:%02d:%02dZ"
    (tm.Unix.tm_year + 1900) (tm.Unix.tm_mon + 1) tm.Unix.tm_mday
    tm.Unix.tm_hour tm.Unix.tm_min tm.Unix.tm_sec

let validate_dispatch_contract payload =
  if contains_nul_byte payload then
    FailClosed {
      reason = "GOSPEL_CONTRACT_VIOLATION: Embedded NUL byte trapped (memchr code -2)";
      error_code = -2;
    }
  else if contains_sql_injection payload then
    FailClosed {
      reason = "GOSPEL_CONTRACT_VIOLATION: Raw unparameterized SQL syntax detected";
      error_code = -3;
    }
  else
    Pass {
      digest = sha256_digest payload;
      timestamp = get_iso_timestamp ();
    }

let reference_oracle_check payload =
  (* Independent reference oracle using byte array traversal *)
  let len = String.length payload in
  let has_nul = ref false in
  for i = 0 to len - 1 do
    if payload.[i] = '\x00' then has_nul := true
  done;
  if !has_nul then FailClosed { reason = "RefOracle: NUL byte"; error_code = -2 }
  else if contains_sql_injection payload then
    FailClosed { reason = "RefOracle: SQL injection"; error_code = -3 }
  else
    FailClosed { reason = "Designated reference disagreement"; error_code = -3 }

let bounded_differential_oracle payload _expected_digest =
  let v1 = validate_dispatch_contract payload in
  let v2 = reference_oracle_check payload in
  match v1, v2 with
  | Pass p1, Pass _ ->
      if String.length p1.digest = 64 then Ok true
      else Error "Digest length mismatch in primary oracle"
  | FailClosed f1, FailClosed f2 ->
      if f1.error_code = f2.error_code then Ok true
      else Error "Error code discrepancy between primary and reference oracle"
  | Pass _, FailClosed f ->
      Error (Printf.sprintf "Discrepancy: Primary passed but reference rejected with code %d" f.error_code)
  | FailClosed f, Pass _ ->
      Error (Printf.sprintf "Discrepancy: Primary rejected with code %d but reference passed" f.error_code)
